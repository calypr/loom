package artifactfs

import (
	"context"
	"fmt"
	"os"
	"strings"
	"testing"
	"time"

	"github.com/calypr/loom/internal/explorer"
)

func TestPreFixExpiryGrowthWitness(t *testing.T) {
	root := t.TempDir()
	store, err := New(root)
	if err != nil {
		t.Fatal(err)
	}
	base := time.Now().UTC().Truncate(time.Second)
	now := base
	store.now = func() time.Time { return now }
	for i := 0; i < 25; i++ {
		record := artifactRecord(t)
		record.ExecutionID = fmt.Sprintf("execution-%02d", i)
		record.IdempotencyKey = fmt.Sprintf("request-%02d", i)
		record.ExpiresAt = base.Add(time.Hour)
		id, err := explorer.ArtifactID(record)
		if err != nil {
			t.Fatal(err)
		}
		record.ID = id
		stage, existing, err := store.Begin(context.Background(), record)
		if err != nil || existing != nil || stage == nil {
			t.Fatalf("begin %d: %v", i, err)
		}
		if _, err := stage.Write([]byte("archive-bytes")); err != nil {
			t.Fatal(err)
		}
		if _, err := stage.Commit(context.Background(), explorer.ArtifactCompletion{ArchiveSHA256: "sha256:witness", Bytes: 13, Rows: 1, Features: 1}); err != nil {
			t.Fatal(err)
		}
	}
	now = base.Add(72 * time.Hour)
	entries, err := os.ReadDir(root)
	if err != nil {
		t.Fatal(err)
	}
	var zips, jsons int
	for _, e := range entries {
		if strings.HasSuffix(e.Name(), ".zip") {
			zips++
		}
		if strings.HasSuffix(e.Name(), ".json") {
			jsons++
		}
	}
	t.Logf("WITNESS 72h past TTL: zips=%d jsons=%d (read paths refuse expired; store offers no removal) — retention unbounded = CONFIRMED growth", zips, jsons)
	if zips != 25 || jsons != 25 {
		t.Fatalf("expected the pre-fix store to retain all 25 expired pairs, saw %d/%d — probe invalid if store gained a sweeper", zips, jsons)
	}
}
